   245→function AnnouncementBar({ onDismiss }: { onDismiss: () => void }) {

   247→  const [dir, setDir] = useState(1);
   248→

   250→    const id = setInterval(() => {
   251→      setDir(1);

   256→
   257→  return (
   258→    <div className="relative z-[52] h-8 bg-[#0A0A0A] flex items-center justify-center overflow-hidden">
   259→      <div className="absolute left-0 right-0 top-1/2 -translate-y-1/2 flex justify-center">
   260→        <div className="h-px w-8 bg-[#C9A96E]/20" />

   264→        <motion.p
   265→          key={idx}
   266→          custom={dir}

   268→          initial="enter"
   269→          animate="center"
   270→          exit="exit"
   271→          className="font-['Inter'] text-[10px] uppercase tracking-[0.22em] text-white/55"

   556→    if (localStorage.getItem('maison-ann-dismissed') === '1') setAnnouncementDismissed(true);
   557→  }, []);
   558→
   559→  const dismissAnnouncement = useCallback(() => {
   560→    setAnnouncementDismissed(true);
   561→    localStorage.setItem('maison-ann-dismissed', '1');
   562→  }, []);
   563→
   564→  /* ---- Scroll detection ---- */
   565→  useMotionValueEvent(scrollY, 'change', (v) => setScrolled(v > 50));
   566→
   567→  /* ---- Body scroll lock ---- */
   568→  useEffect(() => {
   569→    document.body.style.overflow = mobileOpen ? 'hidden' : '';
   570→    return () => { document.body.style.overflow = ''; };
   571→  }, [mobileOpen]);
   572→
   573→  /* ---- Hover handlers with delay ---- */
   574→  const schedule = useCallback((fn: () => void, ms: number) => {
   575→    if (hoverTimer.current) clearTimeout(hoverTimer.current);
   576→    hoverTimer.current = setTimeout(fn, ms);
   577→  }, []);
   578→
   579→  const cancel = useCallback(() => {
   580→    if (hoverTimer.current) { clearTimeout(hoverTimer.current); hoverTimer.current = null; }
   581→  }, []);
   582→
   583→  const enterNav = useCallback((label: string) => schedule(() => setActiveDropdown(label), 280), [schedule]);
   584→  const leaveNav = useCallback(() => schedule(() => setActiveDropdown(null), 180), [schedule]);
   585→  const enterPanel = useCallback(() => cancel(), [cancel]);
   586→  const leavePanel = useCallback(() => schedule(() => setActiveDropdown(null), 180), [schedule]);
   587→
   588→  useEffect(() => () => { if (hoverTimer.current) clearTimeout(hoverTimer.current); }, []);
   589→
   590→  const topOffset = announcementDismissed ? 0 : 32;
   591→
   592→  return (
   593→    <>
   594→      {/* ---- Announcement Bar ---- */}
   595→      {!announcementDismissed && <AnnouncementBar onDismiss={dismissAnnouncement} />}
   596→
   597→      {/* ---- Main Header ---- */}
   598→      <motion.header
   599→        className="fixed left-0 right-0 z-50"
   600→        initial={false}
   601→        animate={{
   602→          backgroundColor: scrolled ? 'rgba(10,10,10,0.88)' : 'rgba(10,10,10,0)',
   603→          backdropFilter: scrolled ? 'blur(28px) saturate(1.4)' : 'blur(0px)',
   604→        }}
   605→        transition={{ duration: 0.55, ease: EASE }}
   606→        style={{ top: topOffset, WebkitBackdropFilter: scrolled ? 'blur(28px) saturate(1.4)' : 'blur(0px)' }}
   607→      >
   608→        {/* Bottom border — fades in on scroll */}
   609→        <motion.div
   610→          className="absolute bottom-0 left-0 right-0 h-px bg-gradient-to-r from-transparent via-white/[0.08] to-transparent"
   611→          initial={{ opacity: 0 }}
   612→          animate={{ opacity: scrolled ? 1 : 0 }}
   613→          transition={{ duration: 0.55, ease: EASE }}
   614→        />
   615→
   616→        <nav className="mx-auto flex h-16 md:h-[72px] max-w-[1400px] items-center justify-between px-5 md:px-10 lg:px-12">
   617→
   618→          {/* ---- Left: Hamburger (mobile) + Nav Links (desktop) ---- */}
   619→          <div className="flex items-center">
   620→            <button
   621→              className="md:hidden relative z-10 p-2 -ml-2 text-white/80 hover:text-white transition-colors duration-300"
   622→              onClick={openMobile}
   623→              aria-label="Open menu"
   624→            >
   625→              <Menu className="h-[18px] w-[18px]" strokeWidth={1.5} />
   626→            </button>
   627→
   628→            <ul className="hidden md:flex items-center">
   629→              {NAV_LINKS.map((link) => {
   630→                const hasDrop = link.hasMegaMenu || (link.subcategories && link.subcategories.length > 0);
   631→                const showDrop = hasDrop && activeDropdown === link.label;
   632→
   633→                return (
   634→                  <li
   635→                    key={link.label}
   636→                    className="relative"
   637→                    onMouseEnter={hasDrop ? () => enterNav(link.label) : undefined}
   638→                    onMouseLeave={hasDrop ? leaveNav : undefined}
   639→                  >
   640→                    <NavLink
   641→                      label={link.label}
   642→                      href={link.href}
   643→                      isActive={!!showDrop}
   644→                    />
   645→
   646→                    {/* Dropdown panels — AnimatePresence wraps conditionally */}
   647→                    {hasDrop && (
   648→                      <div
   649→                        onMouseEnter={enterPanel}
   650→                        onMouseLeave={leavePanel}
   651→                      >
   652→                        <AnimatePresence>
   653→                          {showDrop && (
   654→                            link.hasMegaMenu
   655→                              ? <MegaMenu category={link} />
   656→                              : link.subcategories
   657→                                ? <Dropdown subs={link.subcategories} />
   658→                                : null
   659→                          )}
   660→                        </AnimatePresence>
   661→                      </div>
   662→                    )}
   663→                  </li>
   664→                );
   665→              })}
   666→            </ul>
   667→          </div>
   668→
   669→          {/* ---- Center: Logo ---- */}
   670→          <a
   671→            href="/"
   672→            className="absolute left-1/2 top-1/2 -translate-x-1/2 -translate-y-1/2 select-none z-10"
   673→          >
   674→            <motion.span
   675→              className="font-['Playfair_Display'] text-[26px] md:text-[28px] font-semibold tracking-[0.08em] text-[#C9A96E] inline-block origin-center"
   676→              whileHover={{ letterSpacing: '0.22em' }}
   677→              transition={{ duration: 0.5, ease: EASE }}
   678→            >
   679→              MAISON
   680→            </motion.span>
   681→          </a>
   682→
   683→          {/* ---- Right: Action Icons ---- */}
   684→          <div className="flex items-center gap-0.5 md:gap-1">
   685→            <button
   686→              className="relative z-10 p-2.5 text-white/75 hover:text-white transition-colors duration-300"
   687→              onClick={openSearch}
   688→              aria-label="Search"
   689→            >
   690→              <Search className="h-[18px] w-[18px]" strokeWidth={1.5} />
   691→            </button>
   692→
   693→            <button
   694→              className="relative z-10 p-2.5 text-white/75 hover:text-white transition-colors duration-300"
   695→              aria-label="Wishlist"
   696→            >
   697→              <Heart className="h-[18px] w-[18px]" strokeWidth={1.5} />
   698→              {hydrated && <Badge count={wishCount} />}
   699→            </button>
   700→
   701→            <button
   702→              className="relative z-10 p-2.5 text-white/75 hover:text-white transition-colors duration-300"
   703→              onClick={toggleCart}
   704→              aria-label="Shopping bag"
   705→            >
   706→              <ShoppingBag className="h-[18px] w-[18px]" strokeWidth={1.5} />
   707→              {hydrated && <Badge count={cartCount} />}
   708→            </button>
   709→
   710→            <button
   711→              className="hidden md:block relative z-10 p-2.5 text-white/75 hover:text-white transition-colors duration-300"
   712→              aria-label="Account"
   713→            >
   714→              <User className="h-[18px] w-[18px]" strokeWidth={1.5} />
   715→            </button>
   716→          </div>
   717→        </nav>
   718→      </motion.header>
   719→
   720→      {/* ---- Mobile Menu Overlay ---- */}
   721→      <MobileMenu isOpen={mobileOpen} onClose={closeMobile} />
   722→    </>
   723→  );
   724→}
