   286→   286→/*  Announcement Bar                                                 */
   287→   287→/* ================================================================ */
   288→   288→
   289→   289→function AnnouncementBar() {
   290→   290→  const [dismissed, setDismissed] = useState(false);
   291→   291→  const [[current, direction], setCurrent] = useState([0, 1]);
   292→   292→
   293→   293→  useEffect(() => {
   294→   294→    const stored = localStorage.getItem('maison-announcement-dismissed');
   295→   295→    if (stored === 'true') {
   296→   296→      setDismissed(true);
   297→   297→    }
   298→   298→  }, []);
   299→   299→
   300→   300→  useEffect(() => {
   301→   301→    if (dismissed) return;
   302→   302→    const interval = setInterval(() => {
   303→   303→      setCurrent((prev) => {
   304→   304→        const nextIndex = (prev[0] + 1) % ANNOUNCEMENTS.length;
   305→   305→        return [nextIndex, 1];
   306→   306→      });
   307→   307→    }, 4000);
   308→   308→    return () => clearInterval(interval);
   309→   309→  }, [dismissed]);
   310→   310→
   311→   311→  const handleDismiss = useCallback(() => {
   312→   312→    setDismissed(true);
   313→   313→    localStorage.setItem('maison-announcement-dismissed', 'true');
   314→   314→  }, []);
   315→   315→
   316→   316→  if (dismissed) return null;
   317→   317→
   318→   318→  return (
   319→   319→    <div className="relative h-8 bg-[#0A0A0A] overflow-hidden flex items-center justify-center">
   320→   320→      <AnimatePresence mode="wait" custom={direction}>
   321→   321→        <motion.div
   322→   322→          key={current}
   323→   323→          custom={direction}
   324→   324→          variants={announcementVariants}
   325→   325→          initial="enter"
   326→   326→          animate="center"
   327→   327→          exit="exit"
   328→   328→          className="absolute inset-0 flex items-center justify-center"
   329→   329→        >
   330→   330→          <p className="font-['Inter'] text-[10px] uppercase tracking-[0.2em] text-white/60">
   331→   331→            {ANNOUNCEMENTS[current]}
   332→   332→          </p>
   333→   333→        </motion.div>
   334→   334→      </AnimatePresence>
   335→   335→
   336→   336→      <button
   337→   337→        onClick={handleDismiss}
   338→   338→        className="absolute right-4 top-1/2 -translate-y-1/2 text-white/40 hover:text-white/80 transition-colors duration-300"
   339→   339→        aria-label="Dismiss announcement"
   340→   340→      >
   341→   341→        <X className="h-3 w-3" />
   342→   342→      </button>
   343→   343→    </div>
   344→   344→  );
   345→   345→}

   616→   616→
   617→   617→export default function Navbar() {
   618→   618→  const [scrolled, setScrolled] = useState(false);
   619→   619→  const [activeDropdown, setActiveDropdown] = useState<string | null>(null);
   620→   620→  const hoverTimeoutRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   621→   621→  const { scrollY } = useScroll();
   622→   622→
   623→   623→  const hydrated = useHydrated();
   624→   624→
   625→   625→  const cartItemCount = useCartStore((s) => s.getItemCount());
   626→   626→  const wishlistCount = useWishlistStore((s) => s.getCount());
   627→   627→  const toggleCart = useCartStore((s) => s.toggleCart);
   628→   628→  const openSearch = useUIStore((s) => s.openSearch);
   629→   629→  const mobileMenuOpen = useUIStore((s) => s.mobileMenuOpen);
   630→   630→  const openMobileMenu = useUIStore((s) => s.openMobileMenu);
   631→   631→  const closeMobileMenu = useUIStore((s) => s.closeMobileMenu);
   632→   632→
   633→   633→  /* ---- Scroll detection ---- */
   634→   634→  useMotionValueEvent(scrollY, 'change', (latest) => {
   635→   635→    setScrolled(latest > 40);
   636→   636→  });
   637→   637→
   638→   638→  /* ---- Body scroll lock ---- */
   639→   639→  useEffect(() => {
   640→   640→    if (mobileMenuOpen) {
   641→   641→      document.body.style.overflow = 'hidden';
   642→   642→    } else {
   643→   643→      document.body.style.overflow = '';
   644→   644→    }
   645→   645→    return () => {
   646→   646→      document.body.style.overflow = '';
   647→   647→    };
   648→   648→  }, [mobileMenuOpen]);
   649→   649→
   650→   650→  /* ---- Mega menu hover handlers (with 300ms delay) ---- */
   651→   651→  const handleNavEnter = useCallback((label: string) => {
   652→   652→    if (hoverTimeoutRef.current) {
   653→   653→      clearTimeout(hoverTimeoutRef.current);
   654→   654→      hoverTimeoutRef.current = null;
   655→   655→    }
   656→   656→    hoverTimeoutRef.current = setTimeout(() => {
   657→   657→      setActiveDropdown(label);
   658→   658→    }, 300);
   659→   659→  }, []);
   660→   660→
   661→   661→  const handleNavLeave = useCallback(() => {
   662→   662→    if (hoverTimeoutRef.current) {
   663→   663→      clearTimeout(hoverTimeoutRef.current);
   664→   664→      hoverTimeoutRef.current = null;
   665→   665→    }
   666→   666→    hoverTimeoutRef.current = setTimeout(() => {
   667→   667→      setActiveDropdown(null);
   668→   668→    }, 200);
   669→   669→  }, []);
   670→   670→
   671→   671→  const handleDropdownEnter = useCallback(() => {
   672→   672→    if (hoverTimeoutRef.current) {
   673→   673→      clearTimeout(hoverTimeoutRef.current);
   674→   674→      hoverTimeoutRef.current = null;
   675→   675→    }
   676→   676→  }, []);
   677→   677→
   678→   678→  const handleDropdownLeave = useCallback(() => {
   679→   679→    if (hoverTimeoutRef.current) {
   680→   680→      clearTimeout(hoverTimeoutRef.current);
   681→   681→    }
   682→   682→    hoverTimeoutRef.current = setTimeout(() => {
   683→   683→      setActiveDropdown(null);
   684→   684→    }, 200);
   685→   685→  }, []);
   686→   686→
   687→   687→  /* Cleanup timeout on unmount */
   688→   688→  useEffect(() => {
   689→   689→    return () => {
   690→   690→      if (hoverTimeoutRef.current) {
   691→   691→        clearTimeout(hoverTimeoutRef.current);
   692→   692→      }
   693→   693→    };
   694→   694→  }, []);
   695→   695→
   696→   696→  /* ---- Compute total offset for fixed position ---- */
   697→   697→  const [announcementVisible, setAnnouncementVisible] = useState(true);
   698→   698→
   699→   699→  useEffect(() => {
   700→   700→    const stored = localStorage.getItem('maison-announcement-dismissed');
   701→   701→    if (stored === 'true') {
   702→   702→      setAnnouncementVisible(false);
   703→   703→    }
   704→   704→  }, []);
   705→   705→
   706→   706→  const topOffset = announcementVisible ? 32 : 0;
   707→   707→
   708→   708→  return (
   709→   709→    <>
   710→   710→      {/* Announcement Bar */}
   711→   711→      {announcementVisible && <AnnouncementBar />}
   712→   712→
   713→   713→      {/* Main Navbar Header */}
   714→   714→      <motion.header
   715→   715→        className="fixed left-0 right-0 z-50"
   716→   716→        initial={false}
   717→   717→        animate={{
   718→   718→          backgroundColor: scrolled
   719→   719→            ? 'rgba(10, 10, 10, 0.85)'
   720→   720→            : 'rgba(10, 10, 10, 0)',
   721→   721→          backdropFilter: scrolled ? 'blur(24px)' : 'blur(0px)',
   722→   722→        }}
   723→   723→        transition={{ duration: 0.5, ease: LUXURY_EASE }}
   724→   724→        style={{
   725→   725→          top: topOffset,
   726→   726→          WebkitBackdropFilter: scrolled ? 'blur(24px)' : 'blur(0px)',
   727→   727→        }}
   728→   728→      >
   729→   729→        {/* Subtle bottom border on scroll */}
   730→   730→        <motion.div
   731→   731→          className="absolute bottom-0 left-0 right-0 h-px bg-white/5"
   732→   732→          initial={{ opacity: 0 }}
   733→   733→          animate={{ opacity: scrolled ? 1 : 0 }}
   734→   734→          transition={{ duration: 0.5, ease: LUXURY_EASE }}
   735→   735→        />
   736→   736→
   737→   737→        <nav className="mx-auto flex h-16 md:h-[72px] max-w-7xl items-center justify-between px-5 md:px-8">
   738→   738→          {/* Left: Mobile hamburger + Desktop nav links */}
   739→   739→          <div className="flex items-center gap-1">
   740→   740→            {/* Mobile hamburger */}
   741→   741→            <button
   742→   742→              className="md:hidden text-white/80 hover:text-white transition-colors duration-300 p-2 -ml-2"
   743→   743→              onClick={openMobileMenu}
   744→   744→              aria-label="Open menu"
   745→   745→            >
   746→   746→              <Menu className="h-[18px] w-[18px]" />
   747→   747→            </button>
   748→   748→
   749→   749→            {/* Desktop nav links */}
