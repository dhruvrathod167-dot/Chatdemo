   285→/* ================================================================ */
   286→/*  Announcement Bar                                                 */
   287→/* ================================================================ */
   288→
   289→function AnnouncementBar() {
   290→  const [dismissed, setDismissed] = useState(false);
   291→  const [[current, direction], setCurrent] = useState([0, 1]);
   292→
   293→  useEffect(() => {
   294→    const stored = localStorage.getItem('maison-announcement-dismissed');
   295→    if (stored === 'true') {
   296→      setDismissed(true);
   297→    }
   298→  }, []);
   299→
   300→  useEffect(() => {
   301→    if (dismissed) return;
   302→    const interval = setInterval(() => {
   303→      setCurrent((prev) => {
   304→        const nextIndex = (prev[0] + 1) % ANNOUNCEMENTS.length;
   305→        return [nextIndex, 1];
   306→      });
   307→    }, 4000);
   308→    return () => clearInterval(interval);
   309→  }, [dismissed]);
   310→
   311→  const handleDismiss = useCallback(() => {
   312→    setDismissed(true);
   313→    localStorage.setItem('maison-announcement-dismissed', 'true');
   314→  }, []);
   315→
   316→  if (dismissed) return null;
   317→
   318→  return (
   319→    <div className="relative h-8 bg-[#0A0A0A] overflow-hidden flex items-center justify-center">
   320→      <AnimatePresence mode="wait" custom={direction}>
   321→        <motion.div
   322→          key={current}
   323→          custom={direction}
   324→          variants={announcementVariants}
   325→          initial="enter"
   326→          animate="center"
   327→          exit="exit"
   328→          className="absolute inset-0 flex items-center justify-center"
   329→        >
   330→          <p className="font-['Inter'] text-[10px] uppercase tracking-[0.2em] text-white/60">
   331→            {ANNOUNCEMENTS[current]}
   332→          </p>
   333→        </motion.div>
   334→      </AnimatePresence>
   335→
   336→      <button
   337→        onClick={handleDismiss}
   338→        className="absolute right-4 top-1/2 -translate-y-1/2 text-white/40 hover:text-white/80 transition-colors duration-300"
   339→        aria-label="Dismiss announcement"
   340→      >
   341→        <X className="h-3 w-3" />
   342→      </button>
   343→    </div>
   344→  );
   345→}

   617→export default function Navbar() {

   619→  const [activeDropdown, setActiveDropdown] = useState<string | null>(null);
   620→  const hoverTimeoutRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   621→  const { scrollY } = useScroll();
   622→
   623→  const hydrated = useHydrated();
   624→
   625→  const cartItemCount = useCartStore((s) => s.getItemCount());
   626→  const wishlistCount = useWishlistStore((s) => s.getCount());
   627→  const toggleCart = useCartStore((s) => s.toggleCart);
   628→  const openSearch = useUIStore((s) => s.openSearch);
   629→  const mobileMenuOpen = useUIStore((s) => s.mobileMenuOpen);
   630→  const openMobileMenu = useUIStore((s) => s.openMobileMenu);
   631→  const closeMobileMenu = useUIStore((s) => s.closeMobileMenu);
   632→
   633→  /* ---- Scroll detection ---- */
   634→  useMotionValueEvent(scrollY, 'change', (latest) => {
   635→    setScrolled(latest > 40);
   636→  });
   637→
   638→  /* ---- Body scroll lock ---- */
   639→  useEffect(() => {
   640→    if (mobileMenuOpen) {
   641→      document.body.style.overflow = 'hidden';
   642→    } else {
   643→      document.body.style.overflow = '';
   644→    }
   645→    return () => {
   646→      document.body.style.overflow = '';
   647→    };
   648→  }, [mobileMenuOpen]);
   649→
   650→  /* ---- Mega menu hover handlers (with 300ms delay) ---- */
   651→  const handleNavEnter = useCallback((label: string) => {
   652→    if (hoverTimeoutRef.current) {
   653→      clearTimeout(hoverTimeoutRef.current);
   654→      hoverTimeoutRef.current = null;
   655→    }
   656→    hoverTimeoutRef.current = setTimeout(() => {
   657→      setActiveDropdown(label);
   658→    }, 300);
   659→  }, []);
   660→
   661→  const handleNavLeave = useCallback(() => {
   662→    if (hoverTimeoutRef.current) {
   663→      clearTimeout(hoverTimeoutRef.current);
   664→      hoverTimeoutRef.current = null;
   665→    }
   666→    hoverTimeoutRef.current = setTimeout(() => {
   667→      setActiveDropdown(null);
   668→    }, 200);
   669→  }, []);
   670→
   671→  const handleDropdownEnter = useCallback(() => {
   672→    if (hoverTimeoutRef.current) {
   673→      clearTimeout(hoverTimeoutRef.current);
   674→      hoverTimeoutRef.current = null;
   675→    }
   676→  }, []);
   677→
   678→  const handleDropdownLeave = useCallback(() => {
   679→    if (hoverTimeoutRef.current) {
   680→      clearTimeout(hoverTimeoutRef.current);
   681→    }
   682→    hoverTimeoutRef.current = setTimeout(() => {
   683→      setActiveDropdown(null);
   684→    }, 200);
   685→  }, []);
   686→
   687→  /* Cleanup timeout on unmount */
   688→  useEffect(() => {
   689→    return () => {
   690→      if (hoverTimeoutRef.current) {
   691→        clearTimeout(hoverTimeoutRef.current);
   692→      }
   693→    };
   694→  }, []);
   695→
   696→  /* ---- Compute total offset for fixed position ---- */
   697→  const [announcementVisible, setAnnouncementVisible] = useState(true);
   698→
   699→  useEffect(() => {
   700→    const stored = localStorage.getItem('maison-announcement-dismissed');
   701→    if (stored === 'true') {
   702→      setAnnouncementVisible(false);
   703→    }
   704→  }, []);
   705→
   706→  const topOffset = announcementVisible ? 32 : 0;
   707→
   708→  return (
   709→    <>
   710→      {/* Announcement Bar */}
   711→      {announcementVisible && <AnnouncementBar />}
   712→
   713→      {/* Main Navbar Header */}
   714→      <motion.header
   715→        className="fixed left-0 right-0 z-50"
   716→        initial={false}
   717→        animate={{
   718→          backgroundColor: scrolled
   719→            ? 'rgba(10, 10, 10, 0.85)'
   720→            : 'rgba(10, 10, 10, 0)',
   721→          backdropFilter: scrolled ? 'blur(24px)' : 'blur(0px)',
   722→        }}
   723→        transition={{ duration: 0.5, ease: LUXURY_EASE }}
   724→        style={{
   725→          top: topOffset,
   726→          WebkitBackdropFilter: scrolled ? 'blur(24px)' : 'blur(0px)',
   727→        }}
   728→      >
   729→        {/* Subtle bottom border on scroll */}
   730→        <motion.div
   731→          className="absolute bottom-0 left-0 right-0 h-px bg-white/5"
   732→          initial={{ opacity: 0 }}
   733→          animate={{ opacity: scrolled ? 1 : 0 }}
   734→          transition={{ duration: 0.5, ease: LUXURY_EASE }}
   735→        />
   736→
   737→        <nav className="mx-auto flex h-16 md:h-[72px] max-w-7xl items-center justify-between px-5 md:px-8">
   738→          {/* Left: Mobile hamburger + Desktop nav links */}
   739→          <div className="flex items-center gap-1">
   740→            {/* Mobile hamburger */}
   741→            <button
   742→              className="md:hidden text-white/80 hover:text-white transition-colors duration-300 p-2 -ml-2"
   743→              onClick={openMobileMenu}
   744→              aria-label="Open menu"
   745→            >
   746→              <Menu className="h-[18px] w-[18px]" />
   747→            </button>
   748→
   749→            {/* Desktop nav links */}
   750→            <ul className="hidden md:flex items-center gap-0">
   751→              {NAV_LINKS.map((link) => {
   752→                const hasDropdown =
   753→                  link.hasMegaMenu || (link.subcategories && link.subcategories.length > 0);
   754→                return (
   755→                  <li
   756→                    key={link.label}
   757→                    className="relative"
   758→                    onMouseEnter={
   759→                      hasDropdown ? () => handleNavEnter(link.label) : undefined
   760→                    }
   761→                    onMouseLeave={
   762→                      hasDropdown ? handleNavLeave : undefined
   763→                    }
   764→                  >
   765→                    <NavLink
   766→                      label={link.label}
   767→                      href={link.href}
   768→                      isActive={activeDropdown === link.label}
   769→                      onMouseEnter={
   770→                        hasDropdown ? () => handleNavEnter(link.label) : undefined
   771→                      }
   772→                      onMouseLeave={
   773→                        hasDropdown ? handleNavLeave : undefined
   774→                      }
   775→                    />
   776→
   777→                    {/* Dropdown / Mega menu */}
   778→                    {hasDropdown && activeDropdown === link.label && (
   779→                      <div
   780→                        onMouseEnter={handleDropdownEnter}
   781→                        onMouseLeave={handleDropdownLeave}
   782→                      >
   783→                        <AnimatePresence>
   784→                          {link.hasMegaMenu ? (
   785→                            <MegaMenuPanel category={link} />
   786→                          ) : link.subcategories ? (
   787→                            <SimpleDropdown subcategories={link.subcategories} />
   788→                          ) : null}
   789→                        </AnimatePresence>
   790→                      </div>
   791→                    )}
   792→                  </li>
   793→                );
   794→              })}
   795→            </ul>
   796→          </div>
   797→
   798→          {/* Center: Logo */}
   799→          <a href="/" className="absolute left-1/2 top-1/2 -translate-x-1/2 -translate-y-1/2 select-none">
   800→            <motion.span
   801→              className="font-['Playfair_Display'] text-[28px] font-semibold tracking-wide text-[#C9A96E] inline-block"
   802→              whileHover={{ letterSpacing: '0.25em' }}
   803→              transition={{ duration: 0.45, ease: LUXURY_EASE }}
   804→            >
   805→              MAISON
   806→            </motion.span>
   807→          </a>
   808→
   809→          {/* Right: Action icons */}
   810→          <div className="flex items-center gap-0.5">
   811→            {/* Search */}
   812→            <button
   813→              className="p-2 text-white/80 hover:text-white transition-colors duration-300"
   814→              onClick={openSearch}
   815→              aria-label="Search"
   816→            >
   817→              <Search className="h-[18px] w-[18px]" />
   818→            </button>
   819→
   820→            {/* Wishlist */}
   821→            <button className="relative p-2 text-white/80 hover:text-white transition-colors duration-300" aria-label="Wishlist">
   822→              <Heart className="h-[18px] w-[18px]" />
   823→              {hydrated && <AnimatedBadge count={wishlistCount} />}
   824→            </button>
   825→
   826→            {/* Shopping bag */}
   827→            <button
   828→              className="relative p-2 text-white/80 hover:text-white transition-colors duration-300"
   829→              onClick={toggleCart}
   830→              aria-label="Shopping bag"
   831→            >
   832→              <ShoppingBag className="h-[18px] w-[18px]" />
   833→              {hydrated && <AnimatedBadge count={cartItemCount} />}
   834→            </button>
   835→
   836→            {/* User */}
   837→            <button className="p-2 text-white/80 hover:text-white transition-colors duration-300" aria-label="Account">
   838→              <User className="h-[18px] w-[18px]" />
   839→            </button>
   840→          </div>
   841→        </nav>
   842→      </motion.header>
   843→
   844→      {/* Mobile Menu Overlay */}
   845→      <MobileMenu isOpen={mobileMenuOpen} onClose={closeMobileMenu} />
   846→    </>
   847→  );
   848→}
