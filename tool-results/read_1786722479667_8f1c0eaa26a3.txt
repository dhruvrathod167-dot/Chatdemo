    90→    90→class RegisterView(APIView):
    91→    91→    """
    92→    92→    Register a new user account.
    93→    93→
    94→    94→    Creates a user with the provided email, name, and password.

    96→    96→    """

    98→    98→
    99→    99→    @extend_schema(
   100→   100→        operation_id='auth_register',
   101→   101→        summary='Register New User',
   102→   102→        description=(
   103→   103→            'Create a new user account. Returns the created user data '
   104→   104→            'and JWT tokens for immediate authentication.'
   105→   105→        ),
   106→   106→        tags=['Authentication'],
   107→   107→        request=RegisterSerializer,
   108→   108→        responses={
   109→   109→            201: OpenApiResponse(
   110→   110→                description='User registered successfully.',
   111→   111→                examples=[
   112→   112→                    OpenApiExample(
   113→   113→                        'Created',
   114→   114→                        value={
   115→   115→                            'message': 'Registration successful.',
   116→   116→                            'user': {
   117→   117→                                'id': 1,
   118→   118→                                'email': 'user@example.com',
   119→   119→                                'first_name': 'John',
   120→   120→                                'last_name': 'Doe',
   121→   121→                            },
   122→   122→                            'tokens': {
   123→   123→                                'access': 'eyJ0eXAiOiJKV1Qi...',
   124→   124→                                'refresh': 'eyJ0eXAiOiJKV1Qi...',
   125→   125→                            }
   126→   126→                        }
   127→   127→                    )
   128→   128→                ],
   129→   129→            ),

   131→   131→            409: OpenApiResponse(description='Email already registered.'),
   132→   132→        },
   133→   133→    )
   134→   134→    def post(self, request):
   135→   135→        serializer = RegisterSerializer(data=request.data)
   136→   136→        serializer.is_valid(raise_exception=True)
   137→   137→
   138→   138→        user = serializer.save()
   139→   139→
   140→   140→        # Generate JWT tokens for the new user
   141→   141→        from rest_framework_simplejwt.tokens import RefreshToken
   142→   142→        refresh = RefreshToken.for_user(user)
   143→   143→
   144→   144→        # Send welcome email asynchronously
   145→   145→        try:
   146→   146→            from core.utils import send_welcome_email
   147→   147→            send_welcome_email(user)
   148→   148→        except Exception as e:
   149→   149→            logger.warning(f'Failed to send welcome email to {user.email}: {e}')
   150→   150→
   151→   151→        return Response(
   152→   152→            {
   153→   153→                'message': 'Registration successful.',
   154→   154→                'user': {
   155→   155→                    'id': user.id,
   156→   156→                    'email': user.email,
   157→   157→                    'first_name': user.first_name,
   158→   158→                    'last_name': user.last_name,
   159→   159→                    'full_name': user.get_full_name(),
   160→   160→                    'is_verified': user.is_verified,
   161→   161→                },
   162→   162→                'tokens': {
   163→   163→                    'access': str(refresh.access_token),
   164→   164→                    'refresh': str(refresh),
   165→   165→                },
   166→   166→            },
   167→   167→            status=status.HTTP_201_CREATED,
   168→   168→        )

   733→   733→    def _verify_google_token(self, token):
   734→   734→        """
   735→   735→        Verify a Google OAuth ID token.
   736→   736→
   737→   737→        Uses Google's public key verification via the google-auth library
   738→   738→        if available. Falls back to JWT secret verification for development.
   739→   739→
   740→   740→        Args:
   741→   741→            token: The Google ID token string.
   742→   742→
   743→   743→        Returns:
   744→   744→            dict: Decoded token payload with user info.
   745→   745→
   746→   746→        Raises:
   747→   747→            BadRequestError: If token verification fails.
   748→   748→        """
   749→   749→        try:
   750→   750→            import json
   751→   751→            from jwt import decode, PyJWKClient
   752→   752→
   753→   753→            # Use Google's public keys to verify the token
   754→   754→            jwks_client = PyJWKClient('https://www.googleapis.com/oauth2/v3/certs')
   755→   755→            signing_key = jwks_client.get_signing_key_from_jwt(token)
   756→   756→
   757→   757→            decoded = decode(
   758→   758→                token,
   759→   759→                signing_key.key,
   760→   760→                algorithms=['RS256'],
   761→   761→                audience=getattr(settings, 'GOOGLE_OAUTH_CLIENT_ID', ''),
   762→   762→            )
   763→   763→
   764→   764→            return decoded
   765→   765→
   766→   766→        except ImportError:
   767→   767→            # Fallback: use simple JWT verification with SECRET_KEY
   768→   768→            # This is only for development/testing
   769→   769→            import jwt
   770→   770→            try:
   771→   771→                decoded = jwt.decode(
   772→   772→                    token,
   773→   773→                    settings.SECRET_KEY,
   774→   774→                    algorithms=['HS256'],
   775→   775→                    options={'verify_aud': False},
   776→   776→                )
   777→   777→                return decoded
   778→   778→            except jwt.ExpiredSignatureError:
   779→   779→                raise BadRequestError('Google token has expired.')
   780→   780→            except jwt.InvalidTokenError:
   781→   781→                raise BadRequestError('Invalid Google token.')
   782→   782→
   783→   783→        except Exception as e:
   784→   784→            logger.warning(f'Google token verification failed: {e}')
   785→   785→            raise BadRequestError('Failed to verify Google token. Please try again.')
