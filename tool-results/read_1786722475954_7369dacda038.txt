    90→class RegisterView(APIView):
    91→    """
    92→    Register a new user account.
    93→
    94→    Creates a user with the provided email, name, and password.
    95→    Returns user data and JWT tokens on success.
    96→    """
    97→    permission_classes = [AllowAny]
    98→
    99→    @extend_schema(
   100→        operation_id='auth_register',
   101→        summary='Register New User',
   102→        description=(
   103→            'Create a new user account. Returns the created user data '
   104→            'and JWT tokens for immediate authentication.'
   105→        ),
   106→        tags=['Authentication'],
   107→        request=RegisterSerializer,
   108→        responses={
   109→            201: OpenApiResponse(
   110→                description='User registered successfully.',
   111→                examples=[
   112→                    OpenApiExample(
   113→                        'Created',
   114→                        value={
   115→                            'message': 'Registration successful.',
   116→                            'user': {
   117→                                'id': 1,
   118→                                'email': 'user@example.com',
   119→                                'first_name': 'John',
   120→                                'last_name': 'Doe',
   121→                            },
   122→                            'tokens': {
   123→                                'access': 'eyJ0eXAiOiJKV1Qi...',
   124→                                'refresh': 'eyJ0eXAiOiJKV1Qi...',
   125→                            }
   126→                        }
   127→                    )
   128→                ],
   129→            ),
   130→            400: OpenApiResponse(description='Validation error. Check request body.'),
   131→            409: OpenApiResponse(description='Email already registered.'),
   132→        },
   133→    )
   134→    def post(self, request):
   135→        serializer = RegisterSerializer(data=request.data)
   136→        serializer.is_valid(raise_exception=True)
   137→
   138→        user = serializer.save()
   139→
   140→        # Generate JWT tokens for the new user
   141→        from rest_framework_simplejwt.tokens import RefreshToken
   142→        refresh = RefreshToken.for_user(user)
   143→
   144→        # Send welcome email asynchronously
   145→        try:
   146→            from core.utils import send_welcome_email
   147→            send_welcome_email(user)
   148→        except Exception as e:
   149→            logger.warning(f'Failed to send welcome email to {user.email}: {e}')
   150→
   151→        return Response(
   152→            {
   153→                'message': 'Registration successful.',
   154→                'user': {
   155→                    'id': user.id,
   156→                    'email': user.email,
   157→                    'first_name': user.first_name,
   158→                    'last_name': user.last_name,
   159→                    'full_name': user.get_full_name(),
   160→                    'is_verified': user.is_verified,
   161→                },
   162→                'tokens': {
   163→                    'access': str(refresh.access_token),
   164→                    'refresh': str(refresh),
   165→                },
   166→            },
   167→            status=status.HTTP_201_CREATED,
   168→        )

   306→class UserAddressDetailView(APIView):
   307→    """
   308→    Retrieve, update, or delete a specific user address.
   309→    """
   310→    permission_classes = [IsAuthenticatedAndActive]
   311→
   312→    def _get_address(self, pk, user):
   313→        """
   314→        Fetch a user's address by primary key.
   315→
   316→        Args:
   317→            pk: Address primary key.
   318→            user: The authenticated user.
   319→
   320→        Returns:
   321→            UserAddress instance.
   322→
   323→        Raises:
   324→            NotFoundError: If the address does not exist or doesn't belong to the user.
   325→        """
   326→        try:
   327→            address = UserAddress.objects.get(pk=pk, user=user)
   328→        except UserAddress.DoesNotExist:
   329→            raise NotFoundError('Address not found.')
   330→        return address
   331→
   332→    @extend_schema(
   333→        operation_id='auth_address_retrieve',
   334→        summary='Get User Address',
   335→        description='Retrieve a specific address by ID.',
   336→        tags=['User Addresses'],
   337→        responses={
   338→            200: UserAddressSerializer,
   339→            401: OpenApiResponse(description='Authentication required.'),
   340→            404: OpenApiResponse(description='Address not found.'),
   341→        },
   342→    )
   343→    def get(self, request, pk):
   344→        address = self._get_address(pk, request.user)
   345→        serializer = UserAddressSerializer(address, context={'request': request})
   346→        return Response(serializer.data)
   347→
   348→    @extend_schema(
   349→        operation_id='auth_address_update',
   350→        summary='Update User Address',
   351→        description='Fully update a specific address.',
   352→        tags=['User Addresses'],
   353→        request=UserAddressSerializer,
   354→        responses={
   355→            200: UserAddressSerializer,
   356→            400: OpenApiResponse(description='Validation error.'),
   357→            401: OpenApiResponse(description='Authentication required.'),
   358→            404: OpenApiResponse(description='Address not found.'),
   359→        },
   360→    )
   361→    def put(self, request, pk):
   362→        address = self._get_address(pk, request.user)
   363→        serializer = UserAddressSerializer(
   364→            address,
   365→            data=request.data,
   366→            context={'request': request},
   367→        )
   368→        serializer.is_valid(raise_exception=True)
   369→        serializer.save()
   370→
   371→        return Response(UserAddressSerializer(address, context={'request': request}).data)
   372→
   373→    @extend_schema(
   374→        operation_id='auth_address_partial_update',
   375→        summary='Partially Update User Address',
   376→        description='Partially update a specific address.',
   377→        tags=['User Addresses'],
   378→        request=UserAddressSerializer,
   379→        responses={
   380→            200: UserAddressSerializer,
   381→            400: OpenApiResponse(description='Validation error.'),
   382→            401: OpenApiResponse(description='Authentication required.'),
   383→            404: OpenApiResponse(description='Address not found.'),
   384→        },
   385→    )
   386→    def patch(self, request, pk):
   387→        address = self._get_address(pk, request.user)
   388→        serializer = UserAddressSerializer(
   389→            address,
   390→            data=request.data,
   391→            partial=True,
   392→            context={'request': request},
   393→        )
   394→        serializer.is_valid(raise_exception=True)
   395→        serializer.save()
   396→
   397→        return Response(UserAddressSerializer(address, context={'request': request}).data)
   398→
   399→    @extend_schema(
   400→        operation_id='auth_address_delete',
   401→        summary='Delete User Address',
   402→        description='Delete a specific address.',
   403→        tags=['User Addresses'],
   404→        responses={
   405→            204: OpenApiResponse(description='Address deleted successfully.'),
   406→            401: OpenApiResponse(description='Authentication required.'),
   407→            404: OpenApiResponse(description='Address not found.'),
   408→        },
   409→    )
   410→    def delete(self, request, pk):
   411→        address = self._get_address(pk, request.user)
   412→        address.delete()
   413→        return Response(status=status.HTTP_204_NO_CONTENT)

   466→class PasswordResetRequestView(APIView):
   467→    """
   468→    Request a password reset email.
   469→
   470→    Sends a password reset link to the user's email address.
   471→    Always returns 200 to prevent user enumeration.
   472→    """
   473→    permission_classes = [AllowAny]
   474→
   475→    @extend_schema(
   476→        operation_id='auth_password_reset_request',
   477→        summary='Request Password Reset',
   478→        description=(
   479→            'Send a password reset email to the provided address. '
   480→            'Always returns success to prevent user enumeration.'
   481→        ),
   482→        tags=['Authentication'],
   483→        request=PasswordResetRequestSerializer,
   484→        responses={
   485→            200: OpenApiResponse(
   486→                description='If the email exists, a reset link has been sent.',
   487→                examples=[
   488→                    OpenApiExample(
   489→                        'Success',
   490→                        value={'message': 'If an account with this email exists, a password reset link has been sent.'}
   491→                    )
   492→                ],
   493→            ),
   494→        },
   495→    )
   496→    def post(self, request):
   497→        serializer = PasswordResetRequestSerializer(data=request.data)
   498→        serializer.is_valid(raise_exception=True)
   499→
   500→        email = serializer.validated_data['email']
   501→
   502→        try:
   503→            user = User.objects.get(email=email)
   504→            if user.is_active:
   505→                # Generate password reset token
   506→                uid = urlsafe_base64_encode(force_bytes(user.pk))
   507→                token = default_token_generator.make_token(user)
   508→
   509→                # Build the reset link
   510→                frontend_url = getattr(settings, 'FRONTEND_URL', 'https://maison.com')
   511→                reset_link = f'{frontend_url}/reset-password?uid={uid}&token={token}'
   512→
   513→                from core.utils import send_email_async
   514→
   515→                subject = 'Reset Your MAISON Password'
   516→                body = (
   517→                    f'Dear {user.get_full_name() or "Valued Customer"},\n\n'
   518→                    f'We received a request to reset your password. '
   519→                    f'Click the link below to set a new password:\n\n'
   520→                    f'{reset_link}\n\n'
   521→                    f'This link will expire in 1 hour. '
   522→                    f'If you did not request this, please ignore this email.\n\n'
   523→                    f'With refined regards,\n'
   524→                    f'The MAISON Team'
   525→                )
   526→
   527→                html_body = f"""
   528→                <html>
   529→                <body style="font-family: 'Georgia', serif; color: #333; max-width: 600px; margin: 0 auto; padding: 20px;">
   530→                    <div style="border-bottom: 2px solid #1a1a1a; padding-bottom: 20px; margin-bottom: 30px;">
   531→                        <h1 style="margin: 0; font-size: 28px; letter-spacing: 4px;">MAISON</h1>
   532→                    </div>
   533→                    <h2 style="font-weight: normal; color: #666;">Password Reset</h2>
   534→                    <p>Dear {user.get_full_name() or "Valued Customer"},</p>
   535→                    <p>We received a request to reset your password. Click the button below to set a new password:</p>
   536→                    <div style="text-align: center; margin: 30px 0;">
   537→                        <a href="{reset_link}" style="
   538→                            display: inline-block;
   539→                            background-color: #1a1a1a;
   540→                            color: #fff;
   541→                            padding: 14px 40px;
   542→                            text-decoration: none;
   543→                            letter-spacing: 2px;
   544→                            font-size: 12px;
   545→                        ">RESET PASSWORD</a>
   546→                    </div>
   547→                    <p style="color: #666; font-size: 14px;">This link will expire in 1 hour. If you did not request this, please ignore this email.</p>
   548→                    <div style="border-top: 1px solid #ddd; padding-top: 20px; margin-top: 30px; color: #999; font-size: 12px;">
   549→                        <p style="margin: 0;">With refined regards,<br>The MAISON Team</p>
   550→                    </div>
   551→                </body>
   552→                </html>
   553→                """
   554→
   555→                send_email_async(
   556→                    subject=subject,
   557→                    body=body,
   558→                    to=user.email,
   559→                    html_body=html_body,
   560→                )
   561→
   562→                logger.info(f'Password reset email sent to: {user.email}')
   563→
   564→        except User.DoesNotExist:
   565→            # Do nothing — prevent user enumeration
   566→            pass
   567→
   568→        return Response({
   569→            'message': 'If an account with this email exists, a password reset link has been sent.'
   570→        })

   733→    def _verify_google_token(self, token):
   734→        """
   735→        Verify a Google OAuth ID token.
   736→
   737→        Uses Google's public key verification via the google-auth library
   738→        if available. Falls back to JWT secret verification for development.
   739→
   740→        Args:
   741→            token: The Google ID token string.
   742→
   743→        Returns:
   744→            dict: Decoded token payload with user info.
   745→
   746→        Raises:
   747→            BadRequestError: If token verification fails.
   748→        """
   749→        try:
   750→            import json
   751→            from jwt import decode, PyJWKClient
   752→
   753→            # Use Google's public keys to verify the token
   754→            jwks_client = PyJWKClient('https://www.googleapis.com/oauth2/v3/certs')
   755→            signing_key = jwks_client.get_signing_key_from_jwt(token)
   756→
   757→            decoded = decode(
   758→                token,
   759→                signing_key.key,
   760→                algorithms=['RS256'],
   761→                audience=getattr(settings, 'GOOGLE_OAUTH_CLIENT_ID', ''),
   762→            )
   763→
   764→            return decoded
   765→
   766→        except ImportError:
   767→            # Fallback: use simple JWT verification with SECRET_KEY
   768→            # This is only for development/testing
   769→            import jwt
   770→            try:
   771→                decoded = jwt.decode(
   772→                    token,
   773→                    settings.SECRET_KEY,
   774→                    algorithms=['HS256'],
   775→                    options={'verify_aud': False},
   776→                )
   777→                return decoded
   778→            except jwt.ExpiredSignatureError:
   779→                raise BadRequestError('Google token has expired.')
   780→            except jwt.InvalidTokenError:
   781→                raise BadRequestError('Invalid Google token.')
   782→
   783→        except Exception as e:
   784→            logger.warning(f'Google token verification failed: {e}')
   785→            raise BadRequestError('Failed to verify Google token. Please try again.')
